CREATE DATABASE pawcare

CREATE TABLE user_credentials (
	id INT AUTO_INCREMENT PRIMARY KEY,
    userName varchar(50) NOT NULL UNIQUE,
    email VARCHAR(50) NOT NULL UNIQUE,
    userRole ENUM('User', 'Veterinarian', 'Admin') NOT NULL DEFAULT 'User',
    password VARCHAR(225) NOT NULL,
    created_At TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    isverified TINYINT(1) DEFAULT 0 NOT NULL,
    authType TINYINT(1) DEFAULT 0 NOT NULL
)

CREATE TABLE user_infos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_ID INT NOT NULL,
	firstName VARCHAR(50) NULL,
    middleName VARCHAR(50) NULL,
    lastName VARCHAR(50) NULL,
    suffix VARCHAR(15) NULL,
    phoneNumber VARCHAR(20) NULL,
    houseNum VARCHAR(50) NULL,
    province VARCHAR(50) NULL,
    municipality VARCHAR(50) NULL,
    barangay VARCHAR(50) NULL,
    zipCode VARCHAR(10) NULL,
    profile_Pic LONGBLOB NULL,
    bio TEXT NULL,
    
    CONSTRAINT fk_user
    	FOREIGN KEY (user_ID) REFERENCES user_credentials(id)
    	ON DELETE CASCADE
    	ON UPDATE CASCADE
)

CREATE TABLE user_verification (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  token VARCHAR(64) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES user_credentials(id) ON DELETE CASCADE
);

CREATE TABLE services (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(50) NULL,
    description TEXT NULL,
    image LONGBLOB NULL
);

CREATE TABLE features (
	id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(50) NULL,
    description TEXT NULL,
    icon VARCHAR(100) NULL
);

CREATE TABLE inventory (
    product_ID INT AUTO_INCREMENT PRIMARY KEY,
    item_code VARCHAR(20) NOT NULL UNIQUE,
    photo VARCHAR(255) NULL,
    name VARCHAR(100) NOT NULL,
    item_group VARCHAR(20) NOT NULL,
    date_purchase DATE,
    date_expiration DATE,
    stock INT NULL,
    price DECIMAL(10,2) NULL,
    unit VARCHAR(255) NOT NULL
);

CREATE TABLE online_consultation_table (
	consult_id INT AUTO_INCREMENT PRIMARY KEY NOT NULL,
    channel_consult_ID VARCAHR(50) NOT NULL UNIQUE,
    Owner_name VARCHAR(100) NOT NULL,
    pet_name VARCHAR(100) NOT NULL,
    pet_type VARCHAR(100) NOT NULL,
    payment_proof BLOB NOT NULL,
    concern_text TEXT not NULL,
    type_consult ENUM("urgent", "regular") NOT NULL
fileType ENUM("pdf", "image") NOT NULL
);

CREATE TABLE appointments_tables (
	id_appoint int AUTO_INCREMENT NOT NULL,
    set_time VARCHAR(200) not null,
    set_date DATE not null,
    owner_name VARCHAR(100) not null,
    status ENUM('Approved', 'Pending', 'Declined") NOT NULL DEFAULT('Pending'),
    created_at DATE NOT NULL DEFAULT (CURRENT_DATE),
    PRIMARY KEY(id_appoint)
);

CREATE TABLE orders (
	id_order int AUTO_INCREMENT NOT NULL,
    uid int NOT NULL,
    custormer_name VARCHAR(200) NOT NULL,
    customer_address VARCHAR(255) NOT NULL,
    order_date DATE NOT NULL,
    total FLOAT(10,2) NOT NULL,
    order_status ENUM('Pending','Out Delivery','Shipped','Cancelled') NOT NULL DEFAULT 'Pending',
    landmark TEXT NULL,
    PRIMARY KEY (id_order)
);

CREATE TABLE order_items (
    id_item INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_ID INT NOT NULL,
    product_name VARCHAR(255) NOT NULL,
    quantity INT NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id_order) ON DELETE CASCADE
);

CREATE TABLE notification (
	notify_id int AUTO_INCREMENT NOT NULL,
    UID int not NULL,
    title_notify VARCHAR(200) NOT NULL,
    type_notify VARCHAR(100) NOT NULL,
    details TEXT NOT NULL,
    notify_date DATETIME NOT NULL,
    isRead TINYINT(1) DEFAULT 0 NOT NULL,
    PRIMARY KEY (notify_id),
    
    CONSTRAINT nk_user
    	FOREIGN KEY (UID) REFERENCES user_credentials(id)
    	ON DELETE CASCADE
    	ON UPDATE CASCADE
);     

CREATE TABLE pet_medical_records (
	id_medical_record INT AUTO_INCREMENT NOT NULL,
    owner_name VARCHAR(255) NOT NULL,
    owner_username VARCHAR(80) NOT NULL,
    photo_pet VARCHAR(90) NOT NULL,
    pet_name VARCHAR(100) NOT NULL,
    species ENUM('Dog','Cat') NOT NULL,
    pet_age INT NOT NULL,
    pet_gender ENUM('Male','Female') NOT NULL,
    pet_condition TEXT NULL,
    last_visit DATE NULL,
    diagnosis TEXT NULL,
    PRIMARY KEY (id_medical_record)
);

CREATE TABLE visit_history (
	id_pet_history INT AUTO_INCREMENT NOT NULL,
    id_pet_medical_records INT NOT NULL,
    owner_address VARCHAR(100) NULL,
    owner_email VARCHAR(100) NULL,
    owner_phone VARCHAR(100) NULL,
    day VARCHAR(20) NOT NULL,
    date_visit DATE NOT NULL,
    service_type VARCHAR(90) NOT NULL,
    main_complaint VARCHAR(90) NOT NULL,
    pet_diagnosis VARCHAR(40) NOT NULL,
    treatment_status VARCHAR(100) NOT NULL,
    date_completed_on DATE NOT NULL,
    nursing_issues LONGTEXT NULL,
    care_plan LONGTEXT NULL,
    local_status_check LONGTEXT NULL,
    additional_complaint VARCHAR(255) NULL,
    weight VARCHAR(20) NULL,
    height VARCHAR(20) NULL,
    bmi VARCHAR(20) NULL,
    blood_pressure VARCHAR(20) NULL,
    pulse VARCHAR(20) NULL,
    medications LONGTEXT NULL,
    veterinarian_name VARCHAR(30) NOT NULL,
    PRIMARY KEY (id_pet_history)
);

CREATE TABLE announcements (
  id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(255) NOT NULL,
  content TEXT NOT NULL,
  date_posted DATE NOT NULL,
  expiration_date DATE NOT NULL,
  button_text VARCHAR(50),
  button_link VARCHAR(255)
);
