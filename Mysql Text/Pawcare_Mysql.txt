CREATE DATABASE pawcare

CREATE TABLE user_credentials (
	id INT AUTO_INCREMENT PRIMARY KEY,
    userName varchar(50) NOT NULL UNIQUE,
    email VARCHAR(50) NOT NULL UNIQUE,
    userRole ENUM('User', 'Veterinarian', 'Admin') NOT NULL DEFAULT 'User',
    password VARCHAR(225) NOT NULL,
    created_At TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)

CREATE TABLE user_infos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_ID INT NOT NULL,
	firstName VARCHAR(50) NULL,
    middleName VARCHAR(50) NULL,
    lastName VARCHAR(50) NULL,
    suffix VARCHAR(15) NULL,
    phoneNumber VARCHAR(20) NULL,
    houseNum VARCHAR(50) NULL,
    province VARCHAR(50) NULL,
    municipality VARCHAR(50) NULL,
    barangay VARCHAR(50) NULL,
    zipCode VARCHAR(10) NULL,
    profile_Pic LONGBLOB NULL,
    bio TEXT NULL,
    
    CONSTRAINT fk_user
    	FOREIGN KEY (user_ID) REFERENCES user_credentials(id)
    	ON DELETE CASCADE
    	ON UPDATE CASCADE
)

CREATE TABLE services (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(50) NULL,
    description TEXT NULL,
    image LONGBLOB NULL
);

CREATE TABLE features (
	id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(50) NULL,
    description TEXT NULL,
    icon VARCHAR(100) NULL
);

CREATE TABLE inventory (
    product_ID INT AUTO_INCREMENT PRIMARY KEY,
    item_code VARCHAR(20) NOT NULL UNIQUE,
    photo BLOB NOT NULL,
    name VARCHAR(100) NOT NULL,
    item_group VARCHAR(20) NOT NULL,
    date_purchase DATE,
    date_expiration DATE,
    stock INT NULL,
    price DECIMAL(10,2) NULL,
    unit VARCHAR(255) NOT NULL
);

CREATE TABLE online_consultation_table (
	consult_id INT AUTO_INCREMENT PRIMARY KEY NOT NULL,
    channel_consult_ID VARCAHR(50) NOT NULL UNIQUE,
    Owner_name VARCHAR(100) NOT NULL,
    pet_name VARCHAR(100) NOT NULL,
    pet_type VARCHAR(100) NOT NULL,
    payment_proof BLOB NOT NULL,
    concern_text TEXT not NULL,
    type_consult ENUM("urgent", "regular") NOT NULL
);
